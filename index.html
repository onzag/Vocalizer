Protocol overview 
// Upload a referenced audio file (WAV / MP3 / OGG, < 1 MB)
-> { "action": "upload_audio", "rid": "1", "filename": "rain.wav", "hash": "<sha256>" }
<- { "type": "upload_audio_proceed", "rid": "1", "filename": "rain.wav" }   // then send the raw bytes as ONE binary frame
<- { "type": "upload_audio_skip",    "rid": "1", "filename": "rain.wav" }   // already present, do not send
<- { "type": "upload_audio_done",    "rid": "1", "filename": "rain.wav" }

// Render a scene; files referenced in the JSON resolve to your uploaded files
-> { "action": "render_json", "rid": "2", "output_format": "ogg", "payload": { ...scene... } }
<- { "type": "render_start", "rid": "2", "format": "ogg" }
<- <binary chunks>
<- { "type": "render_done",  "rid": "2", "bytes": 12345 }
